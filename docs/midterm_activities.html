    Explanation/Learnings

    This activity helped me visualize how Java handles memory through its stack and heap. I learned that primitives like int grade live directly on the stack, while objects like Student are stored in the heap, and references like s1, s2, s3 are just pointers on the stack pointing to those heap objects.

    A key insight was realizing that s3 = s2 does not copy the object but copies the reference, so when s3.name = Ana happens, it modifies the same object s2 points to since both share that address in memory. This reinforced how OOP is not just about writing classes but understanding how objects, state, and references actually interact.

    To me, it is like building with Legos: the individual pieces (primitives) are simple and self-contained, but the whole structure (object references) only works when you understand how they connect and share space in memory.

public class Part_2{

    public static void main(String[] args){

        int var_1 = 69;
        int var_2 = 67; 

        System.out.println(69<<2);

        System.out.println(var_1*var_2);                                //  69 * 67              = 4623
        System.out.println((--var_1 - ++var_2) < (++var_1 - --var_2));  // (68 - 68) < (69 - 67) = True
        System.out.println(var_1*(var_2%var_1) == var_2*(var_1/var_2)); //  4623 == 67           = False
        System.out.println( var_1 > var_2 ? var_1 : var_2 );            //  69 > 67 ? 69 : 67    = 69
        System.out.println(var_1<<2);                                   //  1000101 = 100010100
        System.out.println(var_2 | var_1);
    }   
}

import java.util.Scanner;

public class SecureATMSystemMelencio{
	public static void main(String[] args) {
		
		Scanner scan = new Scanner(System.in);
		
		int retries = 0;
		Account account = new Account();
		
		do  {
			System.out.println("Enter PIN: ");
			String input = scan.nextLine();
			
			if (input.equals(account.PIN)) {
				while (true) {
					System.out.println("\033[H\033[2J\033[3J");
					System.out.println("===== ATM MENU ======\n"
				
						+ "1. Check balance\n"
						+ "2. Deposit\n"
						+ "3. Withdraw\n"
						+ "4. Exit\n");
					
					try {
						int selection = scan.nextInt();
	
						switch (selection) {
					
					case 1:
						System.out.println("Current Balance: " + account.bal);
						break;
					case 2:
						System.out.println("Deposit: ");
						float deposit = scan.nextFloat();
						account.Deposit(deposit);
						break;
					case 3:
						System.out.println("Withdraw: ");
						float withdraw = scan.nextFloat();
						account.Withdraw(withdraw);
						break;
					case 4:
						System.out.println("Exited program");
						scan.close();
						return;
						}
					
					}
					catch (Exception e) {
						System.out.println(e);
						scan.nextLine();
					}
				}
			}
			else {
				System.out.println("Wrong PIN! [" + (3 - (++retries)) + " retries left] ");
			}
			
		}
		while (retries < 3);
		
		scan.close();
	}
	
	private static class Account{
		
		String PIN = "Melencio1234";
		float bal;
		
		public Account() {}
		
		public String Withdraw(float amount) {
			
			if (amount > 0) {
				if (amount <= 2000) {
					if (amount <= bal) {
						bal -= amount;
						return "Transaction successful";
					}
					else {
						System.out.println("Withdrawal must be less than or equal to balance");
					}
				}
				else {
					System.out.println("Withdrawal must be 2,000 or below");
				}
			}
			else {
				System.out.println("Withdrawal must be more than 0");
			}
			
			return "/nError!";
		}
		
		public String Deposit(float amount) {
			
			if (amount > 0) {
				
				bal += amount;
				return "Successful transaction";
				
			}

			return "Error!";
		}
	}
}

import java.util.Scanner;

public class StudentPaymentSystemMelencio{
	
	public static void main(String[] args) {
		
		// Initiate scanner
		Scanner scan = new Scanner(System.in);
		
		// Create a new Student class
		Student student = new Student();
		
		student.getInfo(scan);
		
		while (true) {
			System.out.println("===== PAYMENT MENU =====\n" +
							   "1. Pay Tuition\n" +
							   "2. Check Balance\n" +
							   "3. Apply Discount\n" +
							   "4. Exit\n"
			);

			int input = scan.nextInt();

			try {
				switch (input) {
					case 1:
						student.payTuition(scan);
						break;
					case 2:
						student.checkBalance(scan);
						break;
					case 3:
						if (student.canDiscount) student.applyDiscount(scan);
						else System.out.println("Already used discount");
						break;
					case 4:
						student.cleanupExit();
						return;
					default:
						System.out.println("Invalid value! Try again");
						break;
				}
			}
			catch (Exception e) {
				System.out.println(e);
			}
			
			scan.nextLine();
		}
		
	}
	
	private static class Student {
		
		String name;
		String id;
		float tuition;
		int transactionCounter = 0;
		boolean canDiscount = true;
		
		public Student() {}

		public void payTuition(Scanner scan) {
			while (true) {
				System.out.println("Payment amount: ");
				float payment = scan.nextFloat();
				
				if (payment > this.tuition || 0 > payment) {
					System.out.println("Invalid Payment!");
				}
				else {
					System.out.println("Deducted " + payment + " from " + this.tuition + " = New Balance : " + (this.tuition-payment));
					this.tuition -= payment;
					break;
				}
			}
			this.transactionCounter++;
		}


		public void checkBalance(Scanner scan) {
			System.out.println("Remaining Balance: " + this.tuition);
			this.transactionCounter++;
		}


		public void applyDiscount(Scanner scan) {
			while (true) {
				System.out.println("=== Discount Selection\n" +
								   "1. Scholar\n"+
								   "2. Student\n"
				);
				try {
					int selected = scan.nextInt();
					switch (selected) {
						case 1:
							this.canDiscount = false;
							tuition -= (tuition*.20); 
							return;
						case 2:
							System.out.println("Regular Students don't receive discount");
							return;
						default:
							System.out.println("Invalid input");
							break;
					};
				}
				catch (Exception e) {
					System.out.println(e);
				}

				scan.nextLine();
			}
		}


		public void cleanupExit(){
			System.out.println("Student Name: " + this.name);
			System.out.println("Total Transactions: " + this.transactionCounter);
			System.out.println("Final Balance: " + this.tuition);
		}

		
		public void getInfo(Scanner scan) {

			// Request Student Name
			System.out.println("Enter Student Name: ");
			this.name = scan.nextLine();

			// Request Student PIN (String since some can contain letters eg. 2024-01135-MN-0)
			System.out.println("Enter Student PIN: ");
			this.id = scan.nextLine();
					
			// Request Current Tuition
			while(true){
				System.out.println("Enter Total Tuition Fee: ");
				try {
					this.tuition = scan.nextFloat();
					break;
				}
				catch (Exception e){
					System.out.println("Wrong input! " + e);
					scan.nextLine();
				}
			}
			return;
		}
	}
}

import java.util.Scanner;  

public class ExpenseTrackerMelencio {  
    public static void main(String[] args) {  
        Scanner scanner = new Scanner(System.in);  
        float budget = 0;
        float[] totalExpenses = new float[4];  
        
        // Main program loop
        while (true) {
            try {
                // Display program title
                program_title();  
                
                // Get initial budget if not set
                if (budget == 0) budget = get_budget(scanner);  
                
                // Check if expenses exceed budget
                System.out.println(check_balance(scanner, budget, totalExpenses));  
                
                // Show menu options
                System.out.println("---MENU---\n" +
                                  "1. Add expense\n" +
                                  "2. Display budget\n" +
                                  "3. Exit");  
                
                // Process menu selection
                switch (scanner.nextInt()) {
                    case 1:
                        get_expense(scanner, totalExpenses);
                        break;
                    case 2:
                        show_budget(budget, totalExpenses);
                        break;
                    case 3:
                        System.out.println("Exiting program...");
                        return;
                    default:
                        System.out.println("Entry not in menu selection");
                }
            }
            catch (Exception e) {
                System.out.println(e);
            }
            scanner.nextLine();
        }  
    }  

    // Print program title
    static void program_title() {
        System.out.println("           Midterm Activity # 5           ");
        System.out.println("======== PERSONAL EXPENSE TRACKER ========");
    }  

    // Get user budget input
    static float get_budget(Scanner scanner) {
        System.out.println("Enter budget: ");  
        return scanner.nextFloat();
    }  

    // Check if expenses exceed budget
    static String check_balance(Scanner scanner, float budget, float[] totalExpenses) {
        float total = 0;
        for (int i = 0; i < 4; i++) total += totalExpenses[i];  
        return (total > budget) ? "WARNING: Total expense is over budget" : "";
    }  

    // Add new expense to category
    static void get_expense(Scanner scanner, float[] totalExpenses) {  
        while (true) {
            System.out.println("Select expense type:\n" +
                              "1. Food\n" +
                              "2. Transportation\n" +
                              "3. Entertainment\n" +
                              "4. Other Expenses");
            int input = scanner.nextInt();  
            if (0 < input && input < 5) {
                System.out.println("Expense: ");
                totalExpenses[input-1] += scanner.nextFloat();
                return;
            }
            else System.out.println("Invalid expense type");
        }
    }  

    // Display budget and expenses
    static void show_budget(float budget, float[] totalExpenses) {
        float total = 0;
        for (int i = 0; i < 4; i++) total += totalExpenses[i];  
        System.out.println(String.format("Total Budget: %.2f\n" +
                                       "Total Expenses: %.2f\n" +
                                       "Budget: %.2f",
                                       budget, total, (budget - total)));
        System.out.println(String.format("\nFood: %.2f\n" +
                                       "Transportation: %.2f\n" +
                                       "Entertainment: %.2f\n" +
                                       "Other Expenses: %.2f",
                                       totalExpenses[0], totalExpenses[1], 
                                       totalExpenses[2], totalExpenses[3]));
    }
}
